\subsection{Vector fields}

\begin{lem}\label{lem:ballinward}
    Let $M$ be a smooth $n$-manifold with boundary. Let $B$ be a coordinate half ball with corresponding map $\varphi \colon B \to \bH^n$. 
    Then $\frac{\partial}{\partial x^n}$ is a smooth vector field on $B$ which is inward-pointing when restricted to $\boundary M \cap B$.
\end{lem}
\begin{proof}
    The vector field is smooth by Example 8.2 in \cite{Lee2006}. 
    Let $p \in B$. 
    We need to show that $\restrict{\frac{\partial}{\partial x^n}}{p}$ is inward-pointing. 
    By Proposition 5.42 in \cite{Lee2006}, it suffices to show that the $n$-th component of $\restrict{\frac{\partial}{\partial x^n}}{p}$ is positive. 
    Its $n$-th component is 1 so we are done. 
\end{proof}

\begin{lem}[Problem 8-4 in \cite{Lee2006}]
    Let $M$ be a smooth $n$-manifold with boundary. There exists a smooth vector field on $M$, which is inward-pointing when restricted to the boundary.
\end{lem}

\begin{proof}
    Cover $\boundary M$ with a countable number of regular coordinate half-balls $(B_i)_i$. 
    This is possible by Exercise 1.42 in \cite{Lee2006}.
    Let us call the corresponding larger coordinate half-balls $B_i'$ and the corresponding smooth coordinate map $\varphi_i \colon B_i' \to \bH^n$. 
    By Lemma \ref{lem:ballinward}, $\frac{\partial}{\partial x^n}$ is a smooth vector field on $B_i'$ which is inward-pointing when restricted to $\boundary M \cap B$. 
    In particular, $\frac{\partial}{\partial x^n}$ is a smooth vector field along $B_i$. 
    Using Lemma \ref{lem:vecfieldextension}, we can extend these to smooth vector fields $X_i \colon M \to \tanbun{M}$ with support in $B_i'$ and which agree with $\frac{\partial}{\partial x^n}$ on $B_i$. 
    This is again inward-pointing which is easy to see using Proposition 5.42 in \cite{Lee2006} \todo{write Lemma}.
    Now take a partition of unity subordinate to the cover $\{B_i \mid i\} \cup M \setminus \boundary M$. 
    Then do the same thing as in the proof of Lemma \ref{lem:vecfieldextension} and get a global smooth vector field that is inward-pointing on the boundary. 
\end{proof}
